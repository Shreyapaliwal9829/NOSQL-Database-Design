[Comparison notes]
RDBMS: tables/rows, predefined schema, SQL, relationships often use joins.
MongoDB: BSON documents, flexible fields, queries use MongoDB methods, nested data can be embedded.
Both can store the same student information. Scaling depends on the deployment and workload.

PRACTICAL 3 — Orders with embedded customer/products (MongoDB)
============================================================

use CollegeDB
db.Orders.insertMany([
  { Order_ID: 1001, customer: { id: 1, name: "Aarav" }, products: [{ name: "Mouse", price: 500, quantity: 2 }, { name: "Keyboard", price: 900, quantity: 1 }] },
  { Order_ID: 1002, customer: { id: 2, name: "Priya" }, products: [{ name: "Monitor", price: 7500, quantity: 1 }] },
  { Order_ID: 1003, customer: { id: 1, name: "Aarav" }, products: [{ name: "USB Drive", price: 600, quantity: 3 }] },
  { Order_ID: 1004, customer: { id: 3, name: "Rahul" }, products: [{ name: "Webcam", price: 1800, quantity: 1 }, { name: "Headset", price: 1200, quantity: 2 }] },
  { Order_ID: 1005, customer: { id: 4, name: "Neha" }, products: [{ name: "Laptop Stand", price: 1400, quantity: 1 }] }
])

// Orders for a particular customer
db.Orders.find({ "customer.name": "Aarav" })

// Update quantity of Mouse in order 1001
db.Orders.updateOne(
  { Order_ID: 1001, "products.name": "Mouse" },
  { $set: { "products.$.quantity": 3 } }
)

// Calculate total amount per order and display it
db.Orders.aggregate([
  { $unwind: "$products" },
  { $group: {
      _id: "$Order_ID",
      customer: { $first: "$customer.name" },
      totalAmount: { $sum: { $multiply: ["$products.price", "$products.quantity"] } }
  }},
  { $sort: { _id: 1 } }
])

// Orders whose total is above 2000
db.Orders.aggregate([
  { $unwind: "$products" },
  { $group: {
      _id: "$Order_ID",
      customer: { $first: "$customer.name" },
      totalAmount: { $sum: { $multiply: ["$products.price", "$products.quantity"] } }
  }},
  { $match: { totalAmount: { $gt: 2000 } } }
])

PRACTICAL 7 — Neo4j Student-Course graph
============================================================

[Run in Neo4j Browser / Cypher]

CREATE
  (s1:Student {student_id: 1, name: 'Aarav'}),
  (s2:Student {student_id: 2, name: 'Priya'}),
  (s3:Student {student_id: 3, name: 'Rahul'}),
  (s4:Student {student_id: 4, name: 'Neha'}),
  (s5:Student {student_id: 5, name: 'Kabir'}),
  (c1:Course {name: 'BCA'}),
  (c2:Course {name: 'BSc'}),
  (c3:Course {name: 'BCom'}),
  (s1)-[:ENROLLED_IN]->(c1),
  (s2)-[:ENROLLED_IN]->(c1),
  (s3)-[:ENROLLED_IN]->(c2),
  (s4)-[:ENROLLED_IN]->(c1),
  (s5)-[:ENROLLED_IN]->(c3);

MATCH (s:Student) RETURN s;
MATCH (s:Student)-[:ENROLLED_IN]->(c:Course {name: 'BCA'}) RETURN s.name, c.name;
MATCH (s:Student {name: 'Aarav'})-[:ENROLLED_IN]->(c:Course) RETURN c.name;

// Add a relationship
MATCH (s:Student {name: 'Rahul'}), (c:Course {name: 'BCA'})
MERGE (s)-[:ENROLLED_IN]->(c);

// Remove that relationship
MATCH (s:Student {name: 'Rahul'})-[r:ENROLLED_IN]->(c:Course {name: 'BCA'})
DELETE r;

PRACTICAL 10 — MongoDB sharding demonstration
============================================================

[Concept/configuration example — requires a sharded cluster with mongos,
config servers, and shard replica sets already configured.]

sh.enableSharding("ShardDemoDB")
sh.shardCollection("ShardDemoDB.Students", { Student_ID: "hashed" })

use ShardDemoDB
for (let i = 1; i <= 20; i++) {
  db.Students.insertOne({
    Student_ID: i,
    Name: "Student" + i,
    Course: i % 2 === 0 ? "BCA" : "BSc",
    Marks: 60 + (i % 40)
  })
}

db.Students.countDocuments({})
sh.status()

// The shard key is Student_ID hashed, which can distribute inserts across shards.
// Actual distribution must be verified on a configured sharded cluster.

PRACTICAL 11 — MongoDB Primary-Secondary replication
============================================================

[Configuration is performed on each MongoDB server's mongod.conf.
Example replica-set configuration fragment on all members:]

# mongod.conf
replication:
  replSetName: "rsCollege"

[Start each mongod instance with its own data path and port; then connect
to one member with mongosh and initiate the replica set.]

rs.initiate({
  _id: "rsCollege",
  members: [
    { _id: 0, host: "localhost:27017" },
    { _id: 1, host: "localhost:27018" }
  ]
})

rs.status()

// After the replica set elects a primary, insert data on the primary:
use ReplicationDemo
db.Students.insertOne({ Student_ID: 1, Name: "Aarav", Course: "BCA" })

// Check rs.status() and connect directly to the secondary to verify replication.
// A secondary may require read preference secondary for reads.
// If primary fails, an eligible secondary can be elected primary.

PRACTICAL 14 — MapReduce: students per course and average marks
============================================================

[MongoDB mongosh — legacy MapReduce API; deprecated in modern MongoDB.
If unsupported, use aggregation instead.]

use MapReduceDB
db.StudentMarks.insertMany([
  { Student_ID: 1, Name: "Aarav", Course: "BCA", Marks: 86 },
  { Student_ID: 2, Name: "Priya", Course: "BCA", Marks: 91 },
  { Student_ID: 3, Name: "Rahul", Course: "BSc", Marks: 78 },
  { Student_ID: 4, Name: "Neha", Course: "BCA", Marks: 88 },
  { Student_ID: 5, Name: "Kabir", Course: "BSc", Marks: 81 }
])

var mapCourseCount = function () {
  emit(this.Course, { count: 1, totalMarks: this.Marks });
};

var reduceCourseCount = function (course, values) {
  var result = { count: 0, totalMarks: 0 };
  values.forEach(function (v) {
    result.count += v.count;
    result.totalMarks += v.totalMarks;
  });
  return result;
};

db.StudentMarks.mapReduce(
  mapCourseCount,
  reduceCourseCount,
  { out: "CourseMapReduceResults" }
);

db.CourseMapReduceResults.find().forEach(function (doc) {
  printjson({
    Course: doc._id,
    StudentCount: doc.value.count,
    AverageMarks: doc.value.totalMarks / doc.value.count
  });
});

[Modern alternative using aggregation]
db.StudentMarks.aggregate([
  { $group: {
      _id: "$Course",
      StudentCount: { $sum: 1 },
      AverageMarks: { $avg: "$Marks" }
  }}
])
